`lampas-harness`에 아래 기능을 추가합니다.

1. `john-wiki` 루트 경로 설정
2. `AI_CONTEXT.md` 자동 로딩
3. `john-wiki` 문서 검색
4. 문서 읽기
5. 모델에게 tool/function으로 노출
6. 대화 시작 시 장기기억 상태를 명확히 주입
7. “기억에 저장” 아이콘은 선호/규칙 저장용으로 사용하고, 실제 작업 이력은 `john-wiki`에서 조회

## 4. Memory Provider 인터페이스 추가

`lampas_harness/memory/base.py`

```python
from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import list


@dataclass
class MemoryDocument:
    path: str
    title: str
    content: str
    score: float = 0.0


class MemoryProvider(ABC):
    @abstractmethod
    def status(self) -> dict:
        pass

    @abstractmethod
    def read_context(self) -> str:
        pass

    @abstractmethod
    def search(self, query: str, top_k: int = 8) -> list[MemoryDocument]:
        pass

    @abstractmethod
    def read(self, path: str) -> MemoryDocument:
        pass

    @abstractmethod
    def list_pages(self) -> list[str]:
        pass
```

## 5. `john-wiki` Provider 구현

`lampas_harness/memory/john_wiki.py`

```python
import os
import re
from pathlib import Path
from typing import list

from .base import MemoryProvider, MemoryDocument


class JohnWikiMemoryProvider(MemoryProvider):
    def __init__(
        self,
        root: str,
        context_file: str = "AI_CONTEXT.md",
        max_context_chars: int = 12000,
    ):
        self.root = Path(root).expanduser().resolve()
        self.context_file = context_file
        self.max_context_chars = max_context_chars

    def status(self) -> dict:
        return {
            "provider": "john_wiki",
            "root": str(self.root),
            "exists": self.root.exists(),
            "context_file": self.context_file,
            "context_exists": (self.root / self.context_file).exists(),
        }

    def _safe_path(self, path: str) -> Path:
        target = (self.root / path).resolve()

        if not str(target).startswith(str(self.root)):
            raise ValueError("john-wiki root 밖의 파일은 읽을 수 없습니다.")

        return target

    def _markdown_files(self) -> list[Path]:
        if not self.root.exists():
            return []

        return [
            p for p in self.root.rglob("*")
            if p.is_file()
            and p.suffix.lower() in [".md", ".markdown", ".txt"]
            and ".git" not in p.parts
        ]

    def read_context(self) -> str:
        path = self.root / self.context_file

        if not path.exists():
            return (
                "john-wiki 장기기억 provider는 설정되어 있으나 "
                f"{self.context_file} 파일이 없습니다."
            )

        text = path.read_text(encoding="utf-8", errors="ignore")
        return text[: self.max_context_chars]

    def list_pages(self) -> list[str]:
        pages = []

        for p in self._markdown_files():
            pages.append(str(p.relative_to(self.root)))

        return sorted(pages)

    def read(self, path: str) -> MemoryDocument:
        target = self._safe_path(path)

        if not target.exists():
            raise FileNotFoundError(path)

        content = target.read_text(encoding="utf-8", errors="ignore")

        return MemoryDocument(
            path=str(target.relative_to(self.root)),
            title=self._title_from_content(target, content),
            content=content,
            score=1.0,
        )

    def search(self, query: str, top_k: int = 8) -> list[MemoryDocument]:
        query_terms = self._tokenize(query)

        if not query_terms:
            return []

        results = []

        for p in self._markdown_files():
            content = p.read_text(encoding="utf-8", errors="ignore")
            score = self._score(content, query_terms)

            if score > 0:
                rel_path = str(p.relative_to(self.root))
                results.append(
                    MemoryDocument(
                        path=rel_path,
                        title=self._title_from_content(p, content),
                        content=self._excerpt(content, query_terms),
                        score=score,
                    )
                )

        results.sort(key=lambda x: x.score, reverse=True)
        return results[:top_k]

    def _tokenize(self, text: str) -> list[str]:
        tokens = re.findall(r"[가-힣A-Za-z0-9_\-]+", text.lower())
        return [t for t in tokens if len(t) >= 2]

    def _score(self, content: str, terms: list[str]) -> float:
        lower = content.lower()
        score = 0.0

        for term in terms:
            count = lower.count(term)
            if count:
                score += count

        return score

    def _title_from_content(self, path: Path, content: str) -> str:
        for line in content.splitlines():
            line = line.strip()
            if line.startswith("# "):
                return line[2:].strip()

        return path.stem

    def _excerpt(self, content: str, terms: list[str], max_chars: int = 1800) -> str:
        lower = content.lower()

        first_pos = None
        for term in terms:
            pos = lower.find(term)
            if pos != -1:
                first_pos = pos if first_pos is None else min(first_pos, pos)

        if first_pos is None:
            return content[:max_chars]

        start = max(0, first_pos - 500)
        end = min(len(content), start + max_chars)

        return content[start:end]
```

## 6. Provider 초기화 코드

`lampas_harness/memory/__init__.py`

```python
import os

from .john_wiki import JohnWikiMemoryProvider


def get_memory_provider():
    provider = os.getenv("LAMPAS_MEMORY_PROVIDER", "").strip()

    if provider == "john_wiki":
        root = os.getenv("JOHN_WIKI_ROOT")

        if not root:
            raise RuntimeError("JOHN_WIKI_ROOT가 설정되지 않았습니다.")

        context_file = os.getenv("JOHN_WIKI_CONTEXT_FILE", "AI_CONTEXT.md")
        max_context_chars = int(os.getenv("JOHN_WIKI_MAX_CONTEXT_CHARS", "12000"))

        return JohnWikiMemoryProvider(
            root=root,
            context_file=context_file,
            max_context_chars=max_context_chars,
        )

    return None
```

## 7. 모델 컨텍스트에 장기기억 주입

`lampas_harness/context/memory_context.py`

```python
from lampas_harness.memory import get_memory_provider


def build_memory_context() -> str:
    provider = get_memory_provider()

    if provider is None:
        return (
            "장기기억 provider가 설정되어 있지 않습니다. "
            "이전 작업 이력은 현재 대화에 제공된 내용만 기준으로 답해야 합니다."
        )

    status = provider.status()

    if not status.get("exists"):
        return (
            "장기기억 provider는 john-wiki로 설정되어 있으나 "
            f"경로가 존재하지 않습니다: {status.get('root')}"
        )

    context = provider.read_context()

    return f"""
[장기기억 상태]
- provider: john-wiki
- root: {status.get("root")}
- context_file: {status.get("context_file")}
- context_exists: {status.get("context_exists")}

[장기기억 요약]
{context}

[장기기억 사용 규칙]
- 사용자가 과거 작업, 프로젝트 이력, 개인 설정을 물으면 john-wiki를 우선 기준으로 삼는다.
- 단, john-wiki 검색/읽기 결과가 없으면 모른다고 말한다.
- 추측해서 작업 이력을 만들어내지 않는다.
- 필요한 경우 사용자에게 john-wiki 문서명, 목차, 검색어를 요청한다.
"""
```

이제 모델 호출 직전에 system/developer 메시지에 이걸 추가합니다.

예:

```python
from lampas_harness.context.memory_context import build_memory_context


def build_messages(user_messages):
    memory_context = build_memory_context()

    messages = [
        {
            "role": "system",
            "content": "항상 한국어로 응답하라.",
        },
        {
            "role": "developer",
            "content": memory_context,
        },
    ]

    messages.extend(user_messages)
    return messages
```

## 8. `john-wiki` 검색 도구 추가

모델이 필요할 때 직접 검색할 수 있도록 tool을 노출합니다.

`lampas_harness/tools/john_wiki_tools.py`

```python
from lampas_harness.memory import get_memory_provider


def john_wiki_status() -> dict:
    provider = get_memory_provider()

    if provider is None:
        return {
            "ok": False,
            "error": "memory provider가 설정되어 있지 않습니다.",
        }

    return {
        "ok": True,
        "status": provider.status(),
    }


def search_john_wiki(query: str, top_k: int = 8) -> dict:
    provider = get_memory_provider()

    if provider is None:
        return {
            "ok": False,
            "error": "memory provider가 설정되어 있지 않습니다.",
        }

    docs = provider.search(query=query, top_k=top_k)

    return {
        "ok": True,
        "query": query,
        "results": [
            {
                "path": d.path,
                "title": d.title,
                "score": d.score,
                "excerpt": d.content,
            }
            for d in docs
        ],
    }


def read_john_wiki(path: str) -> dict:
    provider = get_memory_provider()

    if provider is None:
        return {
            "ok": False,
            "error": "memory provider가 설정되어 있지 않습니다.",
        }

    try:
        doc = provider.read(path)

        return {
            "ok": True,
            "path": doc.path,
            "title": doc.title,
            "content": doc.content,
        }

    except Exception as e:
        return {
            "ok": False,
            "error": str(e),
        }


def list_john_wiki_pages() -> dict:
    provider = get_memory_provider()

    if provider is None:
        return {
            "ok": False,
            "error": "memory provider가 설정되어 있지 않습니다.",
        }

    return {
        "ok": True,
        "pages": provider.list_pages(),
    }
```

## 11. `john-wiki`에 추가할 `AI_CONTEXT.md`

`john-wiki/AI_CONTEXT.md` 파일을 하나 만드세요.

```markdown
# AI Context

## 사용자 응답 선호
- 항상 한국어로 답변한다.
- 추측보다 확인 가능한 사실을 우선한다.
- 접근할 수 없는 정보는 접근할 수 없다고 명확히 말한다.
- 작업 요약은 "프로젝트 / 완료한 일 / 진행 중인 일 / 다음 행동" 형식을 선호한다.

## 장기기억 구조
- john-wiki는 사용자의 장기기억/개인 위키다.
- 프로젝트 기록, 작업 이력, 의사결정, 참고자료가 john-wiki에 저장된다.
- AI는 과거 작업을 물으면 john-wiki를 우선 기준으로 답해야 한다.

## 주요 프로젝트
- lampas-harness
- john-wiki 장기기억 체계화
- AI 작업 이력 자동 요약

## 현재 개선 목표
- lampas-harness에서 john-wiki를 장기기억 provider로 연결한다.
- 대화 시작 시 AI_CONTEXT.md를 모델 컨텍스트에 주입한다.
- 필요 시 john-wiki 검색/읽기 tool을 사용하게 한다.
```
